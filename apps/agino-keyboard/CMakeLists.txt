cmake_minimum_required(VERSION 3.21)
project(agino-keyboard VERSION 0.1.21 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(AGINO_UNIT_TESTS "Build the unit tests (Qt Test and a private session bus)" ON)
option(AGINO_BUILD_TESTS "Build the integration tests (they need Qt Quick, DTK declarative, GTK, Xvfb, ...)" ON)
option(AGINO_WAYLAND_SHELL "Build the Qt Wayland shell plugin (uses private Qt interfaces)" ON)
option(AGINO_FCITX_LAUNCHERS "Install the launchers and the desktop file for an Fcitx5 with the Agino frontends" ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui DBus Widgets)
find_package(Qt6 REQUIRED COMPONENTS LinguistTools)
add_compile_definitions(AGINO_TRANSLATIONS_DIR="${CMAKE_INSTALL_PREFIX}/share/agino-keyboard/translations")
find_package(Dtk6 REQUIRED COMPONENTS Gui)
find_package(Fcitx5Core REQUIRED)
if(AGINO_WAYLAND_SHELL)
  find_package(Qt6 REQUIRED COMPONENTS WaylandClient)
  add_subdirectory(wayland)
endif()
find_package(X11 REQUIRED COMPONENTS Xi Xtst)
add_library(keyboard-state STATIC src/keyboardstate.cpp)
target_include_directories(keyboard-state PUBLIC src)
target_link_libraries(keyboard-state PUBLIC Qt6::Core)
add_library(fcitx-client STATIC src/fcitxclient.cpp src/virtualkeyboardui.cpp)
target_include_directories(fcitx-client PUBLIC src)
target_link_libraries(fcitx-client PUBLIC keyboard-state Qt6::DBus)
add_library(agino-keyboard-bridge MODULE addon/focusbridge.cpp)
target_link_libraries(agino-keyboard-bridge PRIVATE Fcitx5::Core)
# The only programs allowed to drive the bridge over D-Bus (see trusted() in addon/focusbridge.cpp).
target_compile_definitions(agino-keyboard-bridge PRIVATE AGINO_TRUSTED_CONTROLLER="${CMAKE_INSTALL_PREFIX}/bin/agino-keyboard" AGINO_TRUSTED_UI="${CMAKE_INSTALL_PREFIX}/libexec/agino-keyboard-ui")
configure_file(addon/agino-keyboard.conf.in agino-keyboard.conf @ONLY)
install(TARGETS agino-keyboard-bridge LIBRARY DESTINATION lib/x86_64-linux-gnu/fcitx5)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/agino-keyboard.conf DESTINATION share/fcitx5/addon)
add_library(candidate-rows STATIC src/candidaterows.cpp)
target_include_directories(candidate-rows PUBLIC src)
target_link_libraries(candidate-rows PUBLIC Qt6::Gui)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CANBERRA REQUIRED IMPORTED_TARGET libcanberra)
add_library(key-sound STATIC src/keysound.cpp)
target_include_directories(key-sound PUBLIC src)
target_include_directories(key-sound PRIVATE ${CANBERRA_INCLUDE_DIRS})
target_link_libraries(key-sound PUBLIC Qt6::Core)
target_compile_definitions(key-sound PRIVATE AGINO_KEY_SOUND_PATH="${CMAKE_INSTALL_PREFIX}/share/agino-keyboard/key-click.wav")
install(FILES data/key-click.wav DESTINATION share/agino-keyboard)
# Catalogues (Qt Linguist); the source strings are English, a language without a catalogue shows them as they are.
foreach(lang ar de es fr it ja ko pt_BR ru uk zh_CN zh_HK zh_TW)
  list(APPEND AGINO_TS_FILES translations/agino-keyboard_${lang}.ts)
endforeach()
qt6_add_translation(AGINO_QM_FILES ${AGINO_TS_FILES})
add_custom_target(agino-keyboard-translations ALL DEPENDS ${AGINO_QM_FILES})
install(FILES ${AGINO_QM_FILES} DESTINATION share/agino-keyboard/translations)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
add_library(desktop-adapter STATIC src/desktopadapter.cpp)
target_include_directories(desktop-adapter PUBLIC src)
target_link_libraries(desktop-adapter PUBLIC keyboard-state Qt6::Gui Qt6::DBus X11::X11 X11::Xi X11::Xtst PkgConfig::WAYLAND)
pkg_check_modules(XFIXES REQUIRED IMPORTED_TARGET xcb-xfixes xcb)
if(AGINO_WAYLAND_SHELL)
  target_link_libraries(desktop-adapter PUBLIC Qt6::GuiPrivate)
  target_compile_definitions(desktop-adapter PUBLIC AGINO_WAYLAND_SHELL)
endif()
add_library(panel-platform STATIC src/x11compositor.cpp)
target_include_directories(panel-platform PUBLIC src)
target_link_libraries(panel-platform PRIVATE Qt6::Gui X11::X11 PkgConfig::XFIXES)
add_executable(rasterize-appicon EXCLUDE_FROM_ALL tools/rasterize-appicon.cpp)
target_link_libraries(rasterize-appicon PRIVATE Qt6::Gui)
add_library(idle-platform STATIC src/idleplatform.cpp)
target_include_directories(idle-platform PUBLIC src)
target_link_libraries(idle-platform PUBLIC Qt6::Core Qt6::DBus X11::X11 X11::Xi PkgConfig::WAYLAND)
add_executable(agino-keyboard src/broker.cpp src/preferences.h)
target_link_libraries(agino-keyboard PRIVATE fcitx-client idle-platform)
install(TARGETS agino-keyboard RUNTIME DESTINATION bin)
install(FILES data/agino-keyboard.desktop DESTINATION share/applications)
install(FILES data/agino-keyboard-autostart.desktop DESTINATION /etc/xdg/autostart RENAME agino-keyboard.desktop)
install(FILES data/agino-keyboard.svg DESTINATION share/icons/hicolor/scalable/apps)

set(AGINO_WAYLAND_MODULE "" CACHE FILEPATH "Patched matching Fcitx Wayland frontend")
if(AGINO_WAYLAND_MODULE)
  install(FILES ${AGINO_WAYLAND_MODULE} DESTINATION lib/x86_64-linux-gnu/fcitx5 RENAME libagino-waylandim.so)
  install(FILES addon/aginowaylandim.conf DESTINATION share/fcitx5/addon)
endif()

set(AGINO_KWIN_SOURCE "" CACHE PATH "Exact matching signed Deepin KWin source")
if(AGINO_KWIN_SOURCE)
  find_package(KF6Config REQUIRED)
  find_library(AGINO_KWIN_LIBRARY NAMES kwin.so.5 libkwin.so.5 REQUIRED)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/kwin-sdk/config-kwin.h "#include <kwinconfig.h>\n")
  add_library(agino-keyboard-device MODULE compositor/devicebridge.cpp)
  target_include_directories(agino-keyboard-device PRIVATE ${AGINO_KWIN_SOURCE}/src ${CMAKE_CURRENT_BINARY_DIR}/kwin-sdk)
  target_link_libraries(agino-keyboard-device PRIVATE Qt6::Gui Qt6::DBus KF6::ConfigCore ${AGINO_KWIN_LIBRARY})
  install(TARGETS agino-keyboard-device LIBRARY DESTINATION lib/x86_64-linux-gnu/qt6/plugins/kwin/plugins)
endif()

# Launchers for an Fcitx5 that loads the patched Wayland and X11 frontends of the project (not used by agino).
if(AGINO_FCITX_LAUNCHERS)
  install(FILES data/agino-keyboard-wayland.desktop DESTINATION share/applications)
  install(PROGRAMS data/agino-keyboard-fcitx-wayland data/agino-keyboard-fcitx-x11 DESTINATION bin)
  install(FILES data/org.fcitx.Fcitx5.desktop DESTINATION share/agino-keyboard/autostart)
endif()
set(AGINO_XCB_MODULE "" CACHE FILEPATH "Matching Fcitx X11 registration compatibility module")
if(AGINO_XCB_MODULE)
  install(FILES ${AGINO_XCB_MODULE} DESTINATION lib/agino-keyboard/fcitx5)
endif()

# Production native QtGui frontend. QML is retained only by regression tests.
# The panel of the keyboard: one set of objects for the programs and the tests.
add_library(native-panel OBJECT src/preferences.h native.qrc
 src/native/nativepanel.cpp src/native/nativepanel.h src/native/nativeoutput.cpp src/native/nativecandidaterows.cpp src/native/nativecandidaterows.h src/native/nativepainting.cpp
 src/native/nativesettings.cpp src/native/nativesettings.h src/native/nativeaccessible.cpp)
target_include_directories(native-panel PUBLIC src src/native)
target_link_libraries(native-panel PUBLIC fcitx-client panel-platform Qt6::Gui Dtk6::Gui X11::X11)
add_executable(agino-keyboard-ui src/native-main.cpp)
target_link_libraries(agino-keyboard-ui PRIVATE native-panel candidate-rows desktop-adapter key-sound)
# Keyboard of the login and lock screens: embedded by window id, keys sent as X11 key events (no Fcitx5).
add_executable(agino-keyboard-embed src/embed-main.cpp)
target_link_libraries(agino-keyboard-embed PRIVATE native-panel X11::Xtst)
install(TARGETS agino-keyboard-embed RUNTIME DESTINATION bin)

add_executable(agino-keyboard-dialog EXCLUDE_FROM_ALL src/native/dialog-main.cpp)
target_link_libraries(agino-keyboard-dialog PRIVATE Qt6::Widgets)

add_dependencies(agino-keyboard-ui agino-keyboard-dialog)
install(TARGETS agino-keyboard-ui agino-keyboard-dialog RUNTIME DESTINATION libexec)

if(AGINO_BUILD_TESTS OR AGINO_UNIT_TESTS)
  include(cmake/unit-tests.cmake)
endif()
if(AGINO_BUILD_TESTS)
  include(cmake/tests.cmake)
endif()
